cmake_minimum_required(VERSION 3.24)
project(viewflow_windows_video_compositor LANGUAGES CXX)

enable_testing()
add_executable(viewflow_texture_region_test texture_region_test.cpp)
target_compile_features(viewflow_texture_region_test PRIVATE cxx_std_20)
add_test(NAME texture_region COMMAND viewflow_texture_region_test)
add_executable(viewflow_texture_region_gpu_test texture_region_gpu_test.cpp)
target_compile_features(viewflow_texture_region_gpu_test PRIVATE cxx_std_20)
target_compile_definitions(viewflow_texture_region_gpu_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(viewflow_texture_region_gpu_test PRIVATE d3d11)
add_test(NAME texture_region_gpu COMMAND viewflow_texture_region_gpu_test)

add_library(viewflow_windows_video_compositor STATIC
  video_compositor.cpp
)
target_compile_features(viewflow_windows_video_compositor PUBLIC cxx_std_20)
target_compile_definitions(viewflow_windows_video_compositor PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(viewflow_windows_video_compositor PRIVATE /EHsc /W4 /permissive-)
endif()
target_link_libraries(viewflow_windows_video_compositor PUBLIC mfplat mfuuid ole32 d3d11 dxgi wmcodecdspuuid d3dcompiler)

add_executable(viewflow_windows_video_compositor_test headless_test.cpp)
target_compile_features(viewflow_windows_video_compositor_test PRIVATE cxx_std_20)
target_compile_definitions(viewflow_windows_video_compositor_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(viewflow_windows_video_compositor_test PRIVATE viewflow_windows_video_compositor)

add_executable(viewflow_windows_video_compositor_init_probe init_probe.cpp)
target_compile_features(viewflow_windows_video_compositor_init_probe PRIVATE cxx_std_20)
target_compile_definitions(viewflow_windows_video_compositor_init_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(viewflow_windows_video_compositor_init_probe PRIVATE viewflow_windows_video_compositor dxguid)
