cmake_minimum_required(VERSION 3.24)
project(viewflow_reverse_desktop_observer LANGUAGES C CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(XML "${CMAKE_CURRENT_SOURCE_DIR}/wlr-screencopy-unstable-v1.xml")
add_custom_command(OUTPUT wlr-screencopy-client-protocol.h COMMAND ${WAYLAND_SCANNER} client-header ${XML} "${CMAKE_CURRENT_BINARY_DIR}/wlr-screencopy-client-protocol.h" DEPENDS ${XML})
add_custom_command(OUTPUT wlr-screencopy-protocol.c COMMAND ${WAYLAND_SCANNER} private-code ${XML} "${CMAKE_CURRENT_BINARY_DIR}/wlr-screencopy-protocol.c" DEPENDS ${XML})
add_executable(viewflow_reverse_desktop_observer main.cpp wlr-screencopy-client-protocol.h wlr-screencopy-protocol.c)
target_include_directories(viewflow_reverse_desktop_observer PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_features(viewflow_reverse_desktop_observer PRIVATE cxx_std_20)
target_link_libraries(viewflow_reverse_desktop_observer PRIVATE PkgConfig::WAYLAND)
