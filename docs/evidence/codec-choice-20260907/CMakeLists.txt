cmake_minimum_required(VERSION 3.24)
project(viewflow_codec_bench LANGUAGES CXX)
add_executable(decode decode.cpp)
target_compile_features(decode PRIVATE cxx_std_20)
target_compile_definitions(decode PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(decode PRIVATE mfplat mfuuid ole32 d3d11 dxgi)
add_executable(decode_ffmpeg decode_ffmpeg.cpp)
target_compile_features(decode_ffmpeg PRIVATE cxx_std_20)
target_compile_definitions(decode_ffmpeg PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set(FFROOT "${CMAKE_CURRENT_SOURCE_DIR}/ffmpeg-n8.1-latest-win64-lgpl-shared-8.1")
target_include_directories(decode_ffmpeg PRIVATE "${FFROOT}/include")
target_link_libraries(decode_ffmpeg PRIVATE d3d11 dxgi "${FFROOT}/lib/avcodec.lib" "${FFROOT}/lib/avutil.lib")
