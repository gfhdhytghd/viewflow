cmake_minimum_required(VERSION 3.24)
project(viewflow_portable_input_tests LANGUAGES CXX)
enable_testing()
foreach(name IN ITEMS scroll_units touchpad_routing windows_keys)
    add_executable(${name}_test ../${name}_test.cpp)
    target_compile_features(${name}_test PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${name}_test PRIVATE /UNDEBUG /W4)
    else()
        target_compile_options(${name}_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    endif()
    add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
