cmake_minimum_required(VERSION 3.24)
project(viewflow_linux_window_input LANGUAGES C CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client xkbcommon)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
foreach(PROTOCOL wlr-virtual-pointer-unstable-v1 virtual-keyboard-unstable-v1)
  set(XML "${CMAKE_CURRENT_SOURCE_DIR}/protocols/${PROTOCOL}.xml")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h"
    COMMAND ${WAYLAND_SCANNER} client-header ${XML} "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h" DEPENDS ${XML})
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c"
    COMMAND ${WAYLAND_SCANNER} private-code ${XML} "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c" DEPENDS ${XML})
  list(APPEND PROTOCOL_FILES "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h" "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c")
endforeach()
add_executable(viewflow-linux-window-input main.cpp ${PROTOCOL_FILES})
target_compile_features(viewflow-linux-window-input PRIVATE cxx_std_20)
target_include_directories(viewflow-linux-window-input PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(viewflow-linux-window-input PRIVATE -Wall -Wextra -Werror)
target_link_libraries(viewflow-linux-window-input PRIVATE PkgConfig::WAYLAND)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME input-help COMMAND viewflow-linux-window-input --help)
  add_executable(window-identity-test window_identity_test.cpp)
  target_compile_features(window-identity-test PRIVATE cxx_std_20)
  find_program(LUA_EXECUTABLE NAMES lua lua5.4 REQUIRED)
  add_test(NAME window-identity COMMAND ${LUA_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/window_identity_test.lua $<TARGET_FILE:window-identity-test>)
endif()
