find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
foreach(PROTOCOL xdg-shell viewporter xdg-toplevel-icon-v1 presentation-time)
  if(PROTOCOL STREQUAL "xdg-shell")
    set(XML "${PROTOCOLS}/stable/xdg-shell/xdg-shell.xml")
  elseif(PROTOCOL STREQUAL "presentation-time")
    set(XML "${PROTOCOLS}/stable/presentation-time/presentation-time.xml")
  elseif(PROTOCOL STREQUAL "xdg-toplevel-icon-v1")
    set(XML "${PROTOCOLS}/staging/xdg-toplevel-icon/xdg-toplevel-icon-v1.xml")
  else()
    set(XML "${PROTOCOLS}/stable/viewporter/viewporter.xml")
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h"
    COMMAND ${WAYLAND_SCANNER} client-header ${XML} "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h" DEPENDS ${XML})
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c"
    COMMAND ${WAYLAND_SCANNER} private-code ${XML} "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c" DEPENDS ${XML})
  list(APPEND PROTOCOL_FILES "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h" "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c")
endforeach()
add_executable(viewflow_linux_reverse main.cpp ${PROTOCOL_FILES})
target_include_directories(viewflow_linux_reverse PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(viewflow_linux_reverse PRIVATE viewflow_reverse_decoder PkgConfig::WAYLAND PNG::PNG)

# Independent hardware diagnostic; not built or run by default/CTest. The
# official external UAPI remains under its upstream GPL-2.0-or-later license.
option(VIEWFLOW_BUILD_KVMFR_PROBE "Build explicit opt-in KVMFR EGL import probe" OFF)
set(VIEWFLOW_KVMFR_UAPI_DIR "" CACHE PATH "Looking Glass module directory containing kvmfr.h")
if(VIEWFLOW_BUILD_KVMFR_PROBE)
  if(NOT EXISTS "${VIEWFLOW_KVMFR_UAPI_DIR}/kvmfr.h")
    message(FATAL_ERROR "VIEWFLOW_KVMFR_UAPI_DIR must contain official Looking Glass module/kvmfr.h")
  endif()
  add_executable(viewflow_kvmfr_import_probe kvmfr_import_probe.cpp)
  target_compile_features(viewflow_kvmfr_import_probe PRIVATE cxx_std_20)
  target_compile_options(viewflow_kvmfr_import_probe PRIVATE -Wall -Wextra -Werror)
  target_include_directories(viewflow_kvmfr_import_probe PRIVATE "${VIEWFLOW_KVMFR_UAPI_DIR}")
  target_link_libraries(viewflow_kvmfr_import_probe PRIVATE PkgConfig::GPU)
endif()

option(VIEWFLOW_BUILD_UDMABUF_PROBE "Build opt-in own-memfd udmabuf GPU probe" OFF)
if(VIEWFLOW_BUILD_UDMABUF_PROBE)
  include(CheckIncludeFileCXX)
  check_include_file_cxx("linux/udmabuf.h" VIEWFLOW_HAVE_UDMABUF_HEADER)
  if(NOT VIEWFLOW_HAVE_UDMABUF_HEADER)
    message(FATAL_ERROR "The udmabuf probe requires Linux UAPI linux/udmabuf.h")
  endif()
  add_executable(viewflow_udmabuf_import_probe udmabuf_import_probe.cpp)
  target_compile_features(viewflow_udmabuf_import_probe PRIVATE cxx_std_20)
  target_compile_options(viewflow_udmabuf_import_probe PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(viewflow_udmabuf_import_probe PRIVATE PkgConfig::GPU)
endif()
