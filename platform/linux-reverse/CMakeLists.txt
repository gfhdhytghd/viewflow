cmake_minimum_required(VERSION 3.24)
project(viewflow_linux_reverse LANGUAGES C CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GPU REQUIRED IMPORTED_TARGET libavcodec libavutil egl glesv2)
find_package(CUDAToolkit REQUIRED)
add_library(viewflow_reverse_decoder STATIC gpu_decoder.cpp)
target_compile_features(viewflow_reverse_decoder PUBLIC cxx_std_20)
target_link_libraries(viewflow_reverse_decoder PUBLIC PkgConfig::GPU CUDA::cuda_driver)
target_compile_options(viewflow_reverse_decoder PRIVATE -Wall -Wextra -Werror)
add_executable(viewflow_reverse_decoder_probe decoder_probe.cpp)
target_link_libraries(viewflow_reverse_decoder_probe PRIVATE viewflow_reverse_decoder)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client wayland-egl)
pkg_get_variable(PROTOCOLS wayland-protocols pkgdatadir)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
foreach(PROTOCOL xdg-shell viewporter)
  if(PROTOCOL STREQUAL "xdg-shell")
    set(XML "${PROTOCOLS}/stable/xdg-shell/xdg-shell.xml")
  else()
    set(XML "${PROTOCOLS}/stable/viewporter/viewporter.xml")
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h"
    COMMAND ${WAYLAND_SCANNER} client-header ${XML} "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h" DEPENDS ${XML})
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c"
    COMMAND ${WAYLAND_SCANNER} private-code ${XML} "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c" DEPENDS ${XML})
  list(APPEND PROTOCOL_FILES "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-client-protocol.h" "${CMAKE_CURRENT_BINARY_DIR}/${PROTOCOL}-protocol.c")
endforeach()
add_executable(viewflow_linux_reverse main.cpp ${PROTOCOL_FILES})
target_include_directories(viewflow_linux_reverse PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(viewflow_linux_reverse PRIVATE viewflow_reverse_decoder PkgConfig::WAYLAND)
enable_testing()
add_executable(viewflow_reverse_wire_test ../reverse-common/wire_test.cpp)
target_compile_features(viewflow_reverse_wire_test PRIVATE cxx_std_20)
target_compile_options(viewflow_reverse_wire_test PRIVATE -UNDEBUG)
add_test(NAME reverse_wire COMMAND viewflow_reverse_wire_test)
