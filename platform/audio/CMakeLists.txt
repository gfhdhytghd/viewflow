cmake_minimum_required(VERSION 3.24)
project(viewflow_audio LANGUAGES CXX)
if(APPLE)
  enable_language(OBJCXX)
  add_executable(ViewflowAudio macos_app.mm)
  target_compile_options(ViewflowAudio PRIVATE -fobjc-arc -Wall -Wextra -Werror)
  target_link_libraries(ViewflowAudio PRIVATE "-framework AppKit")
  add_executable(viewflow-audio macos_audio.mm)
  set_target_properties(viewflow-audio PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES
    OSX_DEPLOYMENT_TARGET "14.2")
  target_compile_options(viewflow-audio PRIVATE -fobjc-arc -Wall -Wextra -Werror -Wno-deprecated-declarations)
  target_link_libraries(viewflow-audio PRIVATE "-framework AVFoundation" "-framework CoreAudio" "-framework Foundation")
endif()
if(WIN32)
  add_executable(viewflow-audio windows_audio.cpp)
  target_compile_features(viewflow-audio PRIVATE cxx_std_20)
  target_compile_definitions(viewflow-audio PRIVATE _WIN32_WINNT=0x0A00)
  target_link_libraries(viewflow-audio PRIVATE ole32 mmdevapi)
endif()
