cmake_minimum_required(VERSION 3.24)
project(viewflow_desktop_handoff LANGUAGES CXX)
include(CTest)
add_executable(handoff-core-test core_test.cpp)
target_compile_features(handoff-core-test PRIVATE cxx_std_20)
add_test(NAME handoff-core COMMAND handoff-core-test)
if(WIN32)
  add_executable(viewflow-desktop-handoff windows.cpp)
  target_compile_features(viewflow-desktop-handoff PRIVATE cxx_std_20)
  target_link_libraries(viewflow-desktop-handoff PRIVATE user32)
  add_test(NAME handoff-native-self-test COMMAND viewflow-desktop-handoff --self-test)
elseif(APPLE)
  enable_language(OBJCXX)
  add_executable(viewflow-desktop-handoff macos.mm)
  set_target_properties(viewflow-desktop-handoff PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
  target_compile_options(viewflow-desktop-handoff PRIVATE -fobjc-arc -Wall -Wextra -Werror)
  target_link_libraries(viewflow-desktop-handoff PRIVATE "-framework ApplicationServices" "-framework AppKit")
  add_test(NAME handoff-native-self-test COMMAND viewflow-desktop-handoff --self-test)
endif()
