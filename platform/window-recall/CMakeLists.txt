cmake_minimum_required(VERSION 3.20)
project(viewflow_window_recall LANGUAGES CXX)
include(CTest)
if(WIN32)
  add_executable(viewflow-window-recall windows.cpp)
  target_compile_features(viewflow-window-recall PRIVATE cxx_std_20)
  target_link_libraries(viewflow-window-recall PRIVATE user32 dwmapi)
elseif(APPLE)
  enable_language(OBJCXX)
  add_executable(viewflow-window-recall macos.mm)
  set_target_properties(viewflow-window-recall PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
  target_compile_options(viewflow-window-recall PRIVATE -fobjc-arc -Wall -Wextra -Werror)
  target_link_libraries(viewflow-window-recall PRIVATE "-framework AppKit" "-framework ApplicationServices" "-framework Carbon")
endif()
if(BUILD_TESTING)
  add_executable(viewflow-recall-policy-test policy_test.cpp)
  target_compile_features(viewflow-recall-policy-test PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(viewflow-recall-policy-test PRIVATE /UNDEBUG)
  else()
    target_compile_options(viewflow-recall-policy-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
  endif()
  add_test(NAME recall-policy COMMAND viewflow-recall-policy-test)
endif()
