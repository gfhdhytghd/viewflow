cmake_minimum_required(VERSION 3.27)
project(viewflow-deadline-experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland>=0.56)
add_library(viewflow-deadline-experiment SHARED main.cpp)
target_link_libraries(viewflow-deadline-experiment PRIVATE PkgConfig::HYPRLAND)
target_compile_options(viewflow-deadline-experiment PRIVATE -Wall -Wextra -Wpedantic --no-gnu-unique)
set_target_properties(viewflow-deadline-experiment PROPERTIES PREFIX "")
include(CTest)
add_executable(deadline_test deadline_test.cpp)
target_compile_options(deadline_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME deadline COMMAND deadline_test)
