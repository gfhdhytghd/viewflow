cmake_minimum_required(VERSION 3.24)
project(viewflow_linux_media LANGUAGES C CXX)
set(VIEWFLOW_BUILD_GPU_DMABUF_ENCODER ON CACHE BOOL "Build media encoder" FORCE)
add_subdirectory(../nvenc-encoder encoder)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MEDIA_GPU REQUIRED IMPORTED_TARGET libavcodec libavutil egl glesv2 libdrm)
add_library(viewflow_media_probe_decoder STATIC ../linux-reverse/gpu_decoder.cpp)
target_compile_features(viewflow_media_probe_decoder PUBLIC cxx_std_20)
target_link_libraries(viewflow_media_probe_decoder PUBLIC PkgConfig::MEDIA_GPU ${CMAKE_DL_LIBS})
if(VIEWFLOW_ENABLE_CUDA)
  find_package(CUDAToolkit QUIET)
  if(CUDAToolkit_FOUND)
    target_compile_definitions(viewflow_media_probe_decoder PRIVATE VIEWFLOW_HAVE_CUDA=1)
    target_include_directories(viewflow_media_probe_decoder PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
  endif()
endif()
add_executable(viewflow-media-probe media_probe.cpp)
target_compile_features(viewflow-media-probe PRIVATE cxx_std_20)
target_compile_options(viewflow-media-probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(viewflow-media-probe PRIVATE viewflow-gpu-dmabuf-encoder viewflow_media_probe_decoder)
set_target_properties(viewflow-media-probe PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_BINARY_DIR}/encoder")
