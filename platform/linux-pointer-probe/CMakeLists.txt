cmake_minimum_required(VERSION 3.20)
project(viewflow_linux_pointer_probe LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(viewflow_linux_pointer_probe main.cpp)
target_compile_features(viewflow_linux_pointer_probe PRIVATE cxx_std_20)
target_link_libraries(viewflow_linux_pointer_probe PRIVATE Qt6::Widgets)
target_compile_options(viewflow_linux_pointer_probe PRIVATE -Wall -Wextra -Werror)
add_executable(viewflow_linux_family_geometry_probe family_geometry_probe.cpp)
target_compile_features(viewflow_linux_family_geometry_probe PRIVATE cxx_std_20)
target_link_libraries(viewflow_linux_family_geometry_probe PRIVATE Qt6::Widgets)
target_compile_options(viewflow_linux_family_geometry_probe PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(viewflow_linux_ime_probe_test ime_test.cpp)
  target_compile_features(viewflow_linux_ime_probe_test PRIVATE cxx_std_20)
  target_link_libraries(viewflow_linux_ime_probe_test PRIVATE Qt6::Widgets)
  target_compile_options(viewflow_linux_ime_probe_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME ime_witness COMMAND viewflow_linux_ime_probe_test)
  set_tests_properties(ime_witness PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
endif()
