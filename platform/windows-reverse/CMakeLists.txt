# Compile constant alpha shaders once at build time, including first launch.
file(GLOB _vf_sdk_bins "C:/Program Files (x86)/Windows Kits/10/bin/*/x64")
list(SORT _vf_sdk_bins COMPARE NATURAL ORDER DESCENDING)
find_program(VIEWFLOW_FXC fxc HINTS "$ENV{WindowsSdkVerBinPath}/x64" ${_vf_sdk_bins} REQUIRED)
set(_vf_shader_dir "${CMAKE_CURRENT_BINARY_DIR}/alpha_shaders")
file(MAKE_DIRECTORY "${_vf_shader_dir}")
foreach(_vf_shader IN ITEMS alpha_summary_cs alpha_extract_vs alpha_extract_ps)
    if(_vf_shader STREQUAL "alpha_summary_cs")
        set(_vf_source alpha_summary)
        set(_vf_entry main)
        set(_vf_profile cs_5_0)
    elseif(_vf_shader STREQUAL "alpha_extract_vs")
        set(_vf_source alpha_extract)
        set(_vf_entry vs)
        set(_vf_profile vs_5_0)
    else()
        set(_vf_source alpha_extract)
        set(_vf_entry ps)
        set(_vf_profile ps_5_0)
    endif()
    set(_vf_header "${_vf_shader_dir}/${_vf_shader}.hpp")
    add_custom_command(OUTPUT "${_vf_header}"
        COMMAND "${VIEWFLOW_FXC}" /nologo /O3 /T ${_vf_profile} /E ${_vf_entry}
            /Vn vf_${_vf_shader} /Fh "${_vf_header}"
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${_vf_source}.hlsl"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${_vf_source}.hlsl" VERBATIM)
    target_sources(viewflow_reverse_encoder PRIVATE "${_vf_header}")
endforeach()
target_include_directories(viewflow_reverse_encoder PRIVATE "${_vf_shader_dir}")
