cmake_minimum_required(VERSION 3.24)
project(viewflow_windows_reverse LANGUAGES CXX)
add_library(viewflow_reverse_encoder STATIC hardware_encoder.cpp alpha_plane.cpp)
target_compile_features(viewflow_reverse_encoder PUBLIC cxx_std_20)
target_compile_definitions(viewflow_reverse_encoder PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(viewflow_reverse_encoder PUBLIC d3d11 d3dcompiler dxgi mfplat mf mfuuid ole32 oleaut32 wmcodecdspuuid)
add_executable(viewflow_reverse_encoder_probe encoder_probe.cpp)
target_link_libraries(viewflow_reverse_encoder_probe PRIVATE viewflow_reverse_encoder)
add_subdirectory(../windows-window-capture wgc)
add_executable(viewflow_windows_reverse main.cpp)
target_link_libraries(viewflow_windows_reverse PRIVATE viewflow_reverse_encoder viewflow_windows_window_capture dwmapi windowsapp runtimeobject user32)
enable_testing()
add_executable(viewflow_reverse_wire_test ../reverse-common/wire_test.cpp)
target_compile_features(viewflow_reverse_wire_test PRIVATE cxx_std_20)
target_compile_options(viewflow_reverse_wire_test PRIVATE /UNDEBUG)
add_test(NAME reverse_wire COMMAND viewflow_reverse_wire_test)
add_executable(viewflow_virtual_display virtual_display.cpp)
target_compile_features(viewflow_virtual_display PRIVATE cxx_std_20)
target_compile_definitions(viewflow_virtual_display PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(viewflow_virtual_display PRIVATE setupapi newdev uuid user32)
